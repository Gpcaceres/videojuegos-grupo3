package game

// Board reemplaza a los controles de Windows Forms del proyecto original:
//
//   - Buttons[i]  <-> ListButtons[i].Text   (btnTicTacToe1..9, TabIndex 0..8)
//   - Enabled     <-> grbTicTacToe.Enabled  (GroupBox que contiene los botones)
//   - Events      <-> lo que el usuario "ve": jugadas y MessageBox.Show(...)
//
// La lógica de los controladores trabaja sobre este tablero exactamente igual
// que el original trabajaba sobre los botones; el frontend solo lo dibuja.
type Board struct {
        Buttons [9]string
        Enabled bool
        Events  []Event
}

// Event describe algo que ocurrió en la interfaz, en orden, para que el
// frontend lo pueda animar paso a paso.
type Event struct {
        Type    string `json:"type"`              // "user" | "npc" | "message"
        Index   *int   `json:"index,omitempty"`   // casillero (TabIndex) para user/npc
        Mark    string `json:"mark,omitempty"`    // "X" u "O"
        Text    string `json:"text,omitempty"`    // texto del MessageBox
        Caption string `json:"caption,omitempty"` // título del MessageBox
}

// NewBoard crea el tablero vacío y habilitado.
func NewBoard() *Board {
        return &Board{Enabled: true}
}

// SetText equivale a ListButtons[index].Text = mark.
// who indica quién hizo la jugada ("user" o "npc") para el log de eventos.
func (b *Board) SetText(index int, mark, who string) {
        b.Buttons[index] = mark
        i := index
        b.Events = append(b.Events, Event{Type: who, Index: &i, Mark: mark})
}

// ClearTexts equivale a recorrer ListButtons poniendo Text = "".
func (b *Board) ClearTexts() {
        for i := range b.Buttons {
                b.Buttons[i] = ""
        }
}

// ShowMessage equivale a MessageBox.Show(text, caption).
func (b *Board) ShowMessage(text, caption string) {
        b.Events = append(b.Events, Event{Type: "message", Text: text, Caption: caption})
}

// TakeEvents devuelve y limpia los eventos pendientes.
func (b *Board) TakeEvents() []Event {
        ev := b.Events
        b.Events = nil
        if ev == nil {
                ev = []Event{}
        }
        return ev
}

// Controller es lo que comparten los controladores del profesor
// (MediumAIController, UnbeatableAIController y la versión MiniMax).
// Cada modo debe implementarla.
type Controller interface {
        // InitializeData = InitializeGameData() del original.
        InitializeData()
        InitializeControls(b *Board)
        // PlayUser marca la jugada del usuario; devuelve true si se jugó.
        PlayUser(b *Board, index int) bool
        // PlayNPC hace la jugada del jugador controlado por la computadora (JCC).
        PlayNPC(b *Board)
        // Flag devuelve a quién le toca: "X" u "O".
        Flag() string
        // Weights devuelve mArrayMoves (pesos 0/1/3) para el inspector del frontend.
        // Si el modo no usa pesos (MiniMax), devolver el tablero como 0/1/3.
        Weights() [9]int
}

// Player = enum Player de los controladores del profesor.
type Player int

const (
        PlayerNone Player = 0 // casilla vacía
        PlayerUser Player = 1 // usuario (X)
        PlayerNPC  Player = 2 // JCC (O)
)

// GameState = enum GameState de los controladores del profesor.
type GameState int

const (
        GameOver GameState = 0
        Active   GameState = 1
)

// Constantes de los controladores del profesor (CENTER_INDEX, CORNER_INDICES, SIDE_INDICES).
const CenterIndex = 4

var (
        CornerIndices = [4]int{0, 2, 6, 8}
        SideIndices   = [4]int{1, 3, 5, 7}

        // WinningLines: filas, columnas y diagonales, en el orden del original.
        WinningLines = [8][3]int{
                {0, 1, 2}, {3, 4, 5}, {6, 7, 8},
                {0, 3, 6}, {1, 4, 7}, {2, 5, 8},
                {0, 4, 8}, {2, 4, 6},
        }
)

// FlagForPlayer convierte Player en "X" u "O" (FlagForPlayer del original).
func FlagForPlayer(p Player) string {
        if p == PlayerUser {
                return "X"
        }
        return "O"
}

// Niveles del Proyecto 3 (un modo por variante del enunciado).
const (
        LevelMediumAI     = 1 // MediumAIController            (Tic-Tac-Toe-Versión-IA-Debil)
        LevelUnbeatableAI = 2 // UnbeatableAIController        (Tic-Tac-Toe-Versión-IA-Fuerte)
        LevelMinimax      = 3 // UnbeatableAIController MiniMax (Tic-Tac-Toe-Version-MiniMax)
)

// Mode describe un modo para el selector del frontend.
type Mode struct {
        Level       int    `json:"level"`
        Key         string `json:"key"`
        Name        string `json:"name"`
        Description string `json:"description"`
        Original    string `json:"original"`    // clase del profesor que se porta
        Implemented bool   `json:"implemented"` // false = apartado pendiente para un compañero
        Owner       string `json:"owner"`       // responsable
}

// Modes es el catálogo. Cuando un compañero termine su modo, cambia
// Implemented a true.
var Modes = []Mode{
        {LevelMediumAI, "medium-ai", "MediumAI", "IA intermedia: gana y bloquea con el método numérico, si no, juega al azar.", "MediumAIController", true, "Paul Moreno"},
        {LevelUnbeatableAI, "unbeatable-ai", "UnbeatableAI", "Método numérico heurístico (pesos 1/3/0). Casos 1, 2 y 3.", "UnbeatableAIController", true, "Germán Cáceres"},
        {LevelMinimax, "unbeatable-minmax", "UnbeatableAI · MinMax", "IA imbatible con el algoritmo Minimax.", "UnbeatableAIController (MiniMax)", true, "Diana Guerra"},
}

// FindMode busca un modo por nivel.
func FindMode(level int) (Mode, bool) {
        for _, m := range Modes {
                if m.Level == level {
                        return m, true
                }
        }
        return Mode{}, false
}

import "errors"

var (
        ErrDisabled       = errors.New("el tablero está deshabilitado (la partida terminó)")
        ErrBadIndex       = errors.New("casillero fuera de rango (0..8)")
        ErrUnknownLevel   = errors.New("nivel de juego desconocido")
        ErrNotImplemented = errors.New("este modo todavía no está implementado")
)

// Form es el equivalente a frmTicTacToe del profesor: tiene el tablero
// (los 9 botones + el GroupBox), el controlador del modo elegido y reparte
// los clics. En el original hay una app por modo; aquí se elige con GameLevel.
type Form struct {
        GameLevel     int
        ControlPlayer bool // static bool ControlPlayer del frmTicTacToe
        Board         *Board
        Medium        *MediumAIController
        Unbeatable    *UnbeatableAIController
        Minimax       *MinimaxAIController
}

func NewForm(level int) (*Form, error) {
        f := &Form{
                Board:      NewBoard(),
                Medium:     NewMediumAIController(),
                Unbeatable: NewUnbeatableAIController(),
                Minimax:    NewMinimaxAIController(),
        }
        if err := f.SetLevel(level); err != nil {
                return nil, err
        }
        return f, nil
}

// Controller devuelve el controlador del modo actual.
func (f *Form) Controller() Controller {
        switch f.GameLevel {
        case LevelMediumAI:
                return f.Medium
        case LevelUnbeatableAI:
                return f.Unbeatable
        case LevelMinimax:
                return f.Minimax
        }
        return nil
}

// SetLevel cambia de modo e inicializa el juego.
func (f *Form) SetLevel(level int) error {
        mode, ok := FindMode(level)
        if !ok {
                return ErrUnknownLevel
        }
        if !mode.Implemented {
                return ErrNotImplemented
        }
        f.GameLevel = level
        f.InitializeGame()
        return nil
}

// InitializeGame = frmTicTacToe.InitializeGame (también btnNew).
func (f *Form) InitializeGame() {
        if c := f.Controller(); c != nil {
                c.InitializeData()
                c.InitializeControls(f.Board)
        }
        f.Board.Events = nil
}

// Click = btnTicTacToeN_Click (N = index + 1).
func (f *Form) Click(index int) error {
        if index < 0 || index > 8 {
                return ErrBadIndex
        }
        if !f.Board.Enabled {
                return ErrDisabled
        }
        c := f.Controller()
        if c == nil {
                return ErrUnknownLevel
        }
        f.ControlPlayer = c.PlayUser(f.Board, index)
        if f.ControlPlayer {
                c.PlayNPC(f.Board)
        }
        return nil
}

// CheckWinner checks rows, columns, diagonals, and then a full-board draw.
func (c *MediumAIController) CheckWinner(b *Board) GameState {
        for _, line := range WinningLines {
                cell1 := c.BoardState[line[0]]
                cell2 := c.BoardState[line[1]]
                cell3 := c.BoardState[line[2]]
                if cell1 != PlayerNone && cell1 == cell2 && cell2 == cell3 {
                        b.ShowMessage("\u00a1Gan\u00f3 "+FlagForPlayer(cell1)+"!", "Resultado")
                        c.DisableControls(b)
                        return GameOver
                }
        }

        if c.MoveCount == 9 {
                b.ShowMessage("\u00a1Empate!", "Resultado")
                c.DisableControls(b)
                return GameOver
        }
        return Active
}

func (c *MediumAIController) Weights() [9]int { return c.ArrayMoves }

func (c *MediumAIController) ShowUserMove(b *Board, index int) {
        b.SetText(index, FlagForPlayer(PlayerUser), "user")
        c.BoardState[index] = PlayerUser
        c.MoveCount++
        c.currentPlayerFlag = PlayerNPC
        c.ArrayMoves[index] = 1
}

        if b.Buttons[index] == "" && c.CurrentGameState == Active {
                if c.currentPlayerFlag == PlayerUser {
                        c.ShowUserMove(b, index)
                        c.CurrentGameState = c.CheckWinner(b)

                }
        }
        return false
}

// Port of the professor's UnbeatableAIController.cs and frmTicTacToe.cs.
const (
        minimaxAIPlayer    byte = 'O'
        minimaxHumanPlayer byte = 'X'
        minimaxEmpty       byte = ' '
        // C# int.MinValue and int.MaxValue are signed 32-bit limits.
        minimaxMinValue = -1 << 31
        minimaxMaxValue = 1<<31 - 1
)

type MinimaxAIController struct {
        board       [3][3]byte
        isHumanTurn bool
        isGameOver  bool
}

func NewMinimaxAIController() *MinimaxAIController {
        c := &MinimaxAIController{}
        c.InitializeData()
        return c
}

// InitializeData = ResetBoard (data).
func (c *MinimaxAIController) InitializeData() {
        for i := 0; i < 3; i++ {
                for j := 0; j < 3; j++ {
                        c.board[i][j] = minimaxEmpty
                }
        }
        c.isHumanTurn = true
        c.isGameOver = false
}

// InitializeControls = ResetBoard (buttons).
func (c *MinimaxAIController) InitializeControls(b *Board) {
        b.ClearTexts()
        b.Enabled = true
}

func (c *MinimaxAIController) Flag() string {
        if c.isHumanTurn {
                return "X"
        }
        return "O"
}

// Weights converts characters for the inspector only; Minimax never uses weights.
func (c *MinimaxAIController) Weights() [9]int {
        var weights [9]int
        for i := 0; i < 9; i++ {
                switch c.board[i/3][i%3] {
                case minimaxHumanPlayer:
                        weights[i] = 1
                case minimaxAIPlayer:
                        weights[i] = 3
                }
        }
        return weights
}

// PlayUser = OnButtonClick up to AIMove, dispatched separately by Form.Click.
// The result reports an accepted move, including one that ends the game.
func (c *MinimaxAIController) PlayUser(b *Board, index int) bool {
        if c.isGameOver || !c.isHumanTurn || index < 0 || index >= 9 {
                return false
        }
        row, col := index/3, index%3
        if c.board[row][col] != minimaxEmpty {
                return false
        }
        c.makeMove(b, index, minimaxHumanPlayer)
        if c.checkGameEnd(b) {
                return true
        }
        c.isHumanTurn = false
        return true
}

// PlayNPC = AIMove. The turn guard adapts it to the public Controller API.
func (c *MinimaxAIController) PlayNPC(b *Board) {
        if c.isGameOver || c.isHumanTurn {
                return
        }
        bestMove := c.getBestMove(&c.board)
        if bestMove != -1 {
                c.makeMove(b, bestMove, minimaxAIPlayer)
                c.checkGameEnd(b)
        }
        c.isHumanTurn = true
}

// getBestMove = GetBestMove. Strict > retains the first equally scoring move.
func (c *MinimaxAIController) getBestMove(board *[3][3]byte) int {
        bestScore := minimaxMinValue
        bestMove := -1
        for i := 0; i < 9; i++ {
                row, col := i/3, i%3
                if board[row][col] == minimaxEmpty {
                        board[row][col] = minimaxAIPlayer
                        score := c.minimax(board, 0, false)
                        board[row][col] = minimaxEmpty
                        if score > bestScore {
                                bestScore = score
                                bestMove = i
                        }
                }
        }
        return bestMove
}

// minimax = Minimax, preserving depth scores and row-major traversal.
func (c *MinimaxAIController) minimax(board *[3][3]byte, depth int, isMaximizing bool) int {
        score := c.evaluateBoard(board)
        if score == 10 {
                return score - depth
        }
        if score == -10 {
                return score + depth
        }
        if !c.isMovesLeft(board) {
                return 0
        }
        if isMaximizing {
                best := minimaxMinValue
                for i := 0; i < 3; i++ {
                        for j := 0; j < 3; j++ {
                                if board[i][j] == minimaxEmpty {
                                        board[i][j] = minimaxAIPlayer
                                        best = max(best, c.minimax(board, depth+1, false))
                                        board[i][j] = minimaxEmpty
                                }
                        }
                }
                return best
        }
        best := minimaxMaxValue
        for i := 0; i < 3; i++ {
                for j := 0; j < 3; j++ {
                        if board[i][j] == minimaxEmpty {
                                board[i][j] = minimaxHumanPlayer
                                best = min(best, c.minimax(board, depth+1, true))
                                board[i][j] = minimaxEmpty
                        }
                }
        }
        return best
}

// evaluateBoard = EvaluateBoard: rows, columns, then the two diagonals.
func (c *MinimaxAIController) evaluateBoard(board *[3][3]byte) int {
        for row := 0; row < 3; row++ {
                if board[row][0] == board[row][1] && board[row][1] == board[row][2] {
                        if board[row][0] == minimaxAIPlayer {
                                return 10
                        }
                        if board[row][0] == minimaxHumanPlayer {
                                return -10
                        }
                }
        }
        for col := 0; col < 3; col++ {
                if board[0][col] == board[1][col] && board[1][col] == board[2][col] {
                        if board[0][col] == minimaxAIPlayer {
                                return 10
                        }
                        if board[0][col] == minimaxHumanPlayer {
                                return -10
                        }
                }
        }
        if board[0][0] == board[1][1] && board[1][1] == board[2][2] {
                if board[0][0] == minimaxAIPlayer {
                        return 10
                }
                if board[0][0] == minimaxHumanPlayer {
                        return -10
                }
        }
        if board[0][2] == board[1][1] && board[1][1] == board[2][0] {
                if board[0][2] == minimaxAIPlayer {
                        return 10
                }
                if board[0][2] == minimaxHumanPlayer {
                        return -10
                }
        }
        return 0
}

// isMovesLeft = IsMovesLeft.
func (c *MinimaxAIController) isMovesLeft(board *[3][3]byte) bool {
        for i := 0; i < 3; i++ {
                for j := 0; j < 3; j++ {
                        if board[i][j] == minimaxEmpty {
                                return true
                        }
                }
        }
        return false
}

// makeMove = MakeMove. Occupancy checks replace disabling individual buttons.
func (c *MinimaxAIController) makeMove(b *Board, index int, player byte) {
        c.board[index/3][index%3] = player
        who := "user"
        if player == minimaxAIPlayer {
                who = "npc"
        }
        b.SetText(index, string(player), who)
}

// checkGameEnd = CheckGameEnd, preserving human/AI/draw priority and messages.
func (c *MinimaxAIController) checkGameEnd(b *Board) bool {
        message := ""
        if c.checkWin(minimaxHumanPlayer) {
                message = "You Win!"
        } else if c.checkWin(minimaxAIPlayer) {
                message = "AI Wins!"
        } else if c.isBoardFull() {
                message = "It's a Draw!"
        } else {
                return false
        }
        b.ShowMessage(message, "Tic Tac Toe")
        c.isGameOver = true
        // The web Board represents availability for the whole game.
        b.Enabled = false
        return true
}

// checkWin = CheckWin: interleaved row/column checks, then diagonals.
func (c *MinimaxAIController) checkWin(player byte) bool {
        for i := 0; i < 3; i++ {
                if c.board[i][0] == player && c.board[i][1] == player && c.board[i][2] == player {
                        return true
                }
                if c.board[0][i] == player && c.board[1][i] == player && c.board[2][i] == player {
                        return true
                }
        }
        if c.board[0][0] == player && c.board[1][1] == player && c.board[2][2] == player {
                return true
        }
        if c.board[0][2] == player && c.board[1][1] == player && c.board[2][0] == player {
                return true
        }
        return false
}

// isBoardFull = IsBoardFull.
func (c *MinimaxAIController) isBoardFull() bool {
        for i := 0; i < 3; i++ {
                for j := 0; j < 3; j++ {
                        if c.board[i][j] == minimaxEmpty {
                                return false
                        }
                }
        }
        return true
}

// CheckWinner revisa las 8 líneas en orden y luego el empate (9 jugadas).
func (c *UnbeatableAIController) CheckWinner(b *Board) GameState {
        for _, line := range WinningLines {
                cell1 := c.BoardState[line[0]]
                cell2 := c.BoardState[line[1]]
                cell3 := c.BoardState[line[2]]
                if cell1 != PlayerNone && cell1 == cell2 && cell2 == cell3 {
                        b.ShowMessage("¡Ganó "+FlagForPlayer(cell1)+"!", "Resultado")
                        c.DisableControls(b)
                        return GameOver
                }
        }
        if c.MoveCount == 9 {
                b.ShowMessage("¡Empate!", "Resultado")
                c.DisableControls(b)
                return GameOver
        }
        return Active
}

func (c *UnbeatableAIController) showUserMove(b *Board, index int) {
        b.SetText(index, FlagForPlayer(PlayerUser), "user")
        c.BoardState[index] = PlayerUser
        c.MoveCount++
        c.currentPlayerFlag = PlayerNPC
        c.ArrayMoves[index] = 1
}

// PlayUser: solo si la casilla está vacía y el juego activo; true si se jugó.
func (c *UnbeatableAIController) PlayUser(b *Board, index int) bool {
        if b.Buttons[index] == "" && c.CurrentGameState == Active {
                if c.currentPlayerFlag == PlayerUser {
                        c.showUserMove(b, index)
                        c.CurrentGameState = c.CheckWinner(b)
                        return true
                }
        }
        return false
}

func (c *UnbeatableAIController) generateRandomNPCMove() int {
        for {
                index := rand.Intn(9) // Random.Next(0, 9)
                if c.ArrayMoves[index] == 0 && index < len(c.ArrayMoves) {
                        return index
                }
        }
}

func (c *UnbeatableAIController) showNPCMove(b *Board, index int) {
        b.SetText(index, FlagForPlayer(PlayerNPC), "npc")
        c.BoardState[index] = PlayerNPC
        c.MoveCount++
        c.currentPlayerFlag = PlayerUser
        c.ArrayMoves[index] = 3
}

// pick devuelve el primer casillero vacío de la línea (a, b, d) en ese orden.
// Equivale a: if (m[a]==0) index=a; else if (m[b]==0) index=b; else if (m[d]==0) index=d;
func (c *UnbeatableAIController) pick(a, b, d int) int {
        index := 0
        if c.ArrayMoves[a] == 0 {
                index = a
        } else if c.ArrayMoves[b] == 0 {
                index = b
        } else if c.ArrayMoves[d] == 0 {
                index = d
        }
        c.heuristicSum = c.ArrayMoves[a] + c.ArrayMoves[b] + c.ArrayMoves[d]
        return index
}

func (c *UnbeatableAIController) heuristicNumericalMethod() int {
        m := &c.ArrayMoves
        switch {
        // Suma 6 -> la JCC tiene dos fichas en línea: GANA (Caso 2)
        case m[0]+m[1]+m[2] == 6:
                return c.pick(0, 1, 2)
        case m[3]+m[4]+m[5] == 6:
                return c.pick(3, 4, 5)
        case m[6]+m[7]+m[8] == 6:
                return c.pick(6, 7, 8)
        case m[0]+m[3]+m[6] == 6:
                return c.pick(0, 3, 6)
        case m[1]+m[4]+m[7] == 6:
                return c.pick(1, 4, 7)
        case m[2]+m[5]+m[8] == 6:
                return c.pick(2, 5, 8)
        case m[0]+m[4]+m[8] == 6:
                return c.pick(0, 4, 8)
        case m[2]+m[4]+m[6] == 6:
                return c.pick(2, 4, 6)

        // Suma 2 -> el usuario tiene dos fichas en línea: BLOQUEA (Caso 1)
        case m[0]+m[1]+m[2] == 2:
                return c.pick(0, 1, 2)
        case m[3]+m[4]+m[5] == 2:
                return c.pick(3, 4, 5)
        case m[6]+m[7]+m[8] == 2:
                return c.pick(6, 7, 8)
        case m[0]+m[3]+m[6] == 2:
                return c.pick(0, 3, 6)
        case m[1]+m[4]+m[7] == 2:
                return c.pick(1, 4, 7)
        case m[2]+m[5]+m[8] == 2:
                return c.pick(2, 5, 8)
        case m[0]+m[4]+m[8] == 2:
                return c.pick(0, 4, 8)
        case m[2]+m[4]+m[6] == 2:
                return c.pick(2, 4, 6)

        // Suma 5 en diagonal -> caso trivial: bloqueo en las diagonales (Caso 3)
        case m[0]+m[4]+m[8] == 5:
                var index int
                if m[2] == 0 {
                        index = 2
                } else if m[6] == 0 {
                        index = 6
                } else {
                        index = c.generateRandomNPCMove()
                }
                c.heuristicSum = 5
                return index
        case m[2]+m[4]+m[6] == 5:
                var index int
                if m[0] == 0 {
                        index = 0
                } else if m[8] == 0 {
                        index = 8
                } else {
                        index = c.generateRandomNPCMove()
                }
                c.heuristicSum = 5
                return index

        // Usuario en dos lados adyacentes -> ocupar la esquina entre ellos
        case m[1]+m[3] == 2 && m[0] == 0:
                c.heuristicSum = 2
                return CornerIndices[0]
        case m[1]+m[5] == 2 && m[2] == 0:
                c.heuristicSum = 2
                return CornerIndices[1]
        case m[3]+m[7] == 2 && m[6] == 0:
                c.heuristicSum = 2
                return CornerIndices[2]
        case m[5]+m[7] == 2 && m[8] == 0:
                c.heuristicSum = 2
                return CornerIndices[3]

        default:
                return c.generateRandomNPCMove()
        }
}

// randomFreeAmong repite Random.Next(0, 9) hasta caer en un casillero libre
// que pertenezca a allowed, igual que los do/while del original.
//
// Única diferencia con el original: si no queda NINGÚN casillero libre de ese
// grupo, el original se queda en un bucle infinito (la ventana se congela).
// En un servidor eso bloquearía la petición para siempre, así que en ese caso
// devolvemos ok=false y el llamador sigue con la jugada heurística.
// frozenCases cuenta cuántas veces el original se habría congelado (lo usan las pruebas).
var frozenCases int

func (c *UnbeatableAIController) randomFreeAmong(allowed ...int) (int, bool) {
        anyFree := false
        for _, a := range allowed {
                if c.ArrayMoves[a] == 0 {
                        anyFree = true
                        break
                }
        }
        if !anyFree {
                frozenCases++
                return 0, false
        }
        for {
                index := rand.Intn(9)
                if c.ArrayMoves[index] != 0 {
                        continue
                }
                for _, a := range allowed {
                        if index == a {
                                return index, true
                        }
                }
        }
}

// PlayNPC orquesta el turno de la JCC.
func (c *UnbeatableAIController) PlayNPC(b *Board) {
        if c.CurrentGameState != Active {
                return
        }

        if c.MoveCount == 1 {
                // Primera jugada de la JCC: centro si está libre...
                if c.ArrayMoves[4] == 0 {
                        c.showNPCMove(b, CenterIndex)
                        c.CurrentGameState = c.CheckWinner(b)
                        return
                }
                // ...si no, una esquina aleatoria.
                index, _ := c.randomFreeAmong(CornerIndices[0], CornerIndices[1], CornerIndices[2], CornerIndices[3])
                c.showNPCMove(b, index)
                c.CurrentGameState = c.CheckWinner(b)
                return
        }

        if c.ArrayMoves[4] == 0 {
                c.showNPCMove(b, CenterIndex)
                c.CurrentGameState = c.CheckWinner(b)
                return
        }

        index := c.heuristicNumericalMethod()
        if c.ArrayMoves[4] == 3 && c.heuristicSum == 5 {
                // JCC en el centro y el usuario en esquinas opuestas -> un lado aleatorio.
                // Tal como en el original, la condición compara SIDE_INDICES[0] dos veces
                // (index == SIDE_INDICES[0] || index == SIDE_INDICES[0] || ...[2] || ...[3]),
                // así que el lado 3 (SIDE_INDICES[1]) nunca se elige aquí.
                if side, ok := c.randomFreeAmong(SideIndices[0], SideIndices[0], SideIndices[2], SideIndices[3]); ok {
                        c.showNPCMove(b, side)
                        c.CurrentGameState = c.CheckWinner(b)
                        return
                }
        }
        c.showNPCMove(b, index)
        c.CurrentGameState = c.CheckWinner(b)
}
